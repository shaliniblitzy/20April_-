# =============================================================================
# requirements.txt
# -----------------------------------------------------------------------------
# Purpose : Pinned runtime dependencies for the Python 3 + Flask scaffold.
# Usage   : pip install -r requirements.txt
# Python  : 3.10 or newer (effective minimum imposed by gunicorn 26.0.0;
#           Flask 3.1.x itself supports Python >=3.9).
# Note    : Development and test tooling (pytest, pytest-flask, ruff, mypy)
#           lives in requirements-dev.txt, NOT here. Keep this file lean so
#           production images install only what the running service needs.
#
# Anchored by AAP §0.4.1 (transformation table row), §0.5.1 (key packages),
# §0.5.2 (dependency changes summary), and §0.5.3 (Node→Python mapping is
# documented but NOT applied to this file until the original Node.js
# `package.json` is supplied).
#
# Pinning strategy (AAP §0.5.1):
#   - Flask and gunicorn use exact pins (==) for reproducible production
#     deployments and parity with the AAP's stated contract.
#   - python-dotenv uses a lower-bound pin (>=) per AAP §0.5.1 to allow
#     security patches inside the 1.x series.
#   - Transitive dependencies (Werkzeug, Jinja2, MarkupSafe, ItsDangerous,
#     Click, Blinker) are intentionally NOT hand-pinned here; pip resolves
#     them from Flask 3.1.3's own metadata so security patches flow in
#     automatically. Add explicit pins only if a lock file is introduced.
# =============================================================================

# -----------------------------------------------------------------------------
# Web framework
# -----------------------------------------------------------------------------
# Flask 3.1.3 (released 2026-02-19) is the latest stable release on PyPI and
# declares `Requires-Python >=3.9`. Installing Flask transitively pulls in:
#   - Werkzeug    (WSGI utilities, request/response objects, dev server)
#   - Jinja2      (template engine; available for future server-rendered HTML)
#   - MarkupSafe  (auto-escaping primitives backing Jinja2)
#   - ItsDangerous (cryptographic signing of session cookies)
#   - Click       (the `flask` CLI)
#   - Blinker     (signals; required dependency from Flask 2.3+)
# These are resolved automatically by pip from Flask's distribution metadata,
# so they are deliberately omitted below.
Flask==3.1.3

# -----------------------------------------------------------------------------
# Production WSGI server
# -----------------------------------------------------------------------------
# gunicorn 26.0.0 (released 2026-05-05) is the latest stable release on PyPI
# and declares `Requires-Python >=3.10`. It uses a pre-fork worker model with
# configurable worker classes (sync, gthread, gevent). The canonical launch
# command for this scaffold is:
#
#     gunicorn wsgi:app
#
# Typical I/O-bound starting point for tuning:
#
#     gunicorn --workers $((2*$(nproc)+1)) --worker-class gthread \
#              --threads 4 --bind 0.0.0.0:5000 wsgi:app
gunicorn==26.0.0

# -----------------------------------------------------------------------------
# Environment file loading
# -----------------------------------------------------------------------------
# python-dotenv reads a `.env` file at process start and exports its contents
# into `os.environ`. The `flask` CLI auto-detects python-dotenv when it is
# installed and loads `.env` and `.flaskenv` for local development, so this
# package is part of the runtime contract (NOT a dev-only dependency). The
# `.env.example` file in this repository documents every variable the
# application reads (FLASK_APP, FLASK_CONFIG, SECRET_KEY, HOST, PORT,
# LOG_LEVEL).
python-dotenv>=1.0.1
